CREATE DATABASE IF NOT EXISTS bdInteriorGo
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE bdInteriorGo;


-- =========================================================
-- 1. USUARIO
-- =========================================================

CREATE TABLE USUARIO (
    userId INT AUTO_INCREMENT PRIMARY KEY,

    nome VARCHAR(100) NOT NULL,

    email VARCHAR(150) NOT NULL UNIQUE,

    senha VARCHAR(255) NOT NULL,

    cpf CHAR(11) NOT NULL UNIQUE,

    fotoPerfil VARCHAR(255),

    dataCadastro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    statusConta ENUM(
        'Ativa',
        'Inativa',
        'Bloqueada'
    ) NOT NULL DEFAULT 'Ativa'
);


-- =========================================================
-- 2. TELEFONE
-- =========================================================

CREATE TABLE TELEFONE (
    telefoneId INT AUTO_INCREMENT PRIMARY KEY,

    userId INT NOT NULL,

    numeroTelefone VARCHAR(20) NOT NULL,

    CONSTRAINT fk_telefone_usuario
        FOREIGN KEY (userId)
        REFERENCES USUARIO(userId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 3. ENDERECO
-- =========================================================

CREATE TABLE ENDERECO (
    enderecoId INT AUTO_INCREMENT PRIMARY KEY,

    userId INT NOT NULL,

    logradouro VARCHAR(150) NOT NULL,

    numero VARCHAR(20),

    bairro VARCHAR(100),

    cidade VARCHAR(100),

    estado CHAR(2),

    cep VARCHAR(9),

    complemento VARCHAR(100),

    CONSTRAINT fk_endereco_usuario
        FOREIGN KEY (userId)
        REFERENCES USUARIO(userId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 4. ENDERECO_FAVORITO
-- =========================================================

CREATE TABLE ENDERECO_FAVORITO (
    enderecoFavoritoId INT AUTO_INCREMENT PRIMARY KEY,

    userId INT NOT NULL,

    nome VARCHAR(100) NOT NULL,

    endereco VARCHAR(255) NOT NULL,

    tipo ENUM(
        'Casa',
        'Trabalho',
        'Escola',
        'Hospital',
        'Outro'
    ) NOT NULL DEFAULT 'Outro',

    CONSTRAINT fk_endereco_favorito_usuario
        FOREIGN KEY (userId)
        REFERENCES USUARIO(userId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 5. PASSAGEIRO
-- =========================================================

CREATE TABLE PASSAGEIRO (
    passageiroId INT AUTO_INCREMENT PRIMARY KEY,

    userId INT NOT NULL UNIQUE,

    status ENUM(
        'Ativo',
        'Inativo'
    ) NOT NULL DEFAULT 'Ativo',

    CONSTRAINT fk_passageiro_usuario
        FOREIGN KEY (userId)
        REFERENCES USUARIO(userId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 6. MOTORISTA
-- =========================================================

CREATE TABLE MOTORISTA (
    motoristaId INT AUTO_INCREMENT PRIMARY KEY,

    userId INT NOT NULL UNIQUE,

    cnh VARCHAR(20) NOT NULL,

    validadeCNH DATE NOT NULL,

    avaliacaoMedia DECIMAL(3,2) NOT NULL DEFAULT 0.00,

    statusMotorista ENUM(
        'Pendente',
        'Aprovado',
        'Bloqueado'
    ) NOT NULL DEFAULT 'Pendente',

    CONSTRAINT fk_motorista_usuario
        FOREIGN KEY (userId)
        REFERENCES USUARIO(userId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 7. ADMINISTRADOR
-- =========================================================

CREATE TABLE ADMINISTRADOR (
    administradorId INT AUTO_INCREMENT PRIMARY KEY,

    userId INT NOT NULL UNIQUE,

    nivelAcesso ENUM(
        'Master',
        'Operador'
    ) NOT NULL DEFAULT 'Operador',

    CONSTRAINT fk_administrador_usuario
        FOREIGN KEY (userId)
        REFERENCES USUARIO(userId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 8. VEICULO
-- =========================================================

CREATE TABLE VEICULO (
    veiculoId INT AUTO_INCREMENT PRIMARY KEY,

    motoristaId INT NOT NULL,

    marca VARCHAR(50) NOT NULL,

    modelo VARCHAR(50) NOT NULL,

    placa VARCHAR(10) NOT NULL UNIQUE,

    cor VARCHAR(30),

    anoFabricacao YEAR,

    anoModelo YEAR,

    categoria ENUM(
        'Economico',
        'Comfort',
        'SUV',
        'Moto'
    ) NOT NULL DEFAULT 'Economico',

    CONSTRAINT fk_veiculo_motorista
        FOREIGN KEY (motoristaId)
        REFERENCES MOTORISTA(motoristaId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 9. PONTO_INTERESSE
-- =========================================================
-- Cadastrado pelo administrador.
-- Pode ser utilizado como ponto de embarque.

CREATE TABLE PONTO_INTERESSE (
    pontoInteresseId INT AUTO_INCREMENT PRIMARY KEY,

    administradorId INT NULL,

    nome VARCHAR(100) NOT NULL,

    descricao VARCHAR(255),

    categoria ENUM(
        'Hospital',
        'Escola',
        'Faculdade',
        'Comercio',
        'Transporte',
        'Praca',
        'Prefeitura',
        'Saude',
        'Outro'
    ) NOT NULL DEFAULT 'Outro',

    logradouro VARCHAR(150),

    numero VARCHAR(20),

    bairro VARCHAR(100),

    cidade VARCHAR(100),

    estado CHAR(2),

    cep VARCHAR(9),

    latitude DECIMAL(10,8) NOT NULL,

    longitude DECIMAL(11,8) NOT NULL,

    pontoEmbarque BOOLEAN NOT NULL DEFAULT FALSE,

    status ENUM(
        'Ativo',
        'Inativo'
    ) NOT NULL DEFAULT 'Ativo',

    CONSTRAINT fk_ponto_interesse_administrador
        FOREIGN KEY (administradorId)
        REFERENCES ADMINISTRADOR(administradorId)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);


-- =========================================================
-- 10. PONTO_DE_ENCONTRO
-- =========================================================
-- Cadastrado pelo próprio passageiro.

CREATE TABLE PONTO_DE_ENCONTRO (
    pontoEncontroId INT AUTO_INCREMENT PRIMARY KEY,

    passageiroId INT NOT NULL,

    nome VARCHAR(100) NOT NULL,

    latitude DECIMAL(10,8) NOT NULL,

    longitude DECIMAL(11,8) NOT NULL,

    descricao VARCHAR(255),

    cidade VARCHAR(100) NOT NULL,

    CONSTRAINT fk_ponto_encontro_passageiro
        FOREIGN KEY (passageiroId)
        REFERENCES PASSAGEIRO(passageiroId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 11. EVENTO
-- =========================================================
-- Cadastrado pelo administrador.

CREATE TABLE EVENTO (
    eventoId INT AUTO_INCREMENT PRIMARY KEY,

    administradorId INT NULL,

    nome VARCHAR(150) NOT NULL,

    descricao TEXT,

    local VARCHAR(255) NOT NULL,

    dataInicio DATETIME NOT NULL,

    dataFim DATETIME NOT NULL,

    categoria VARCHAR(100),

    cidade VARCHAR(100) NOT NULL,

    ativo BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_evento_administrador
        FOREIGN KEY (administradorId)
        REFERENCES ADMINISTRADOR(administradorId)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);


-- =========================================================
-- 12. CORRIDA_AGENDADA
-- =========================================================

CREATE TABLE CORRIDA_AGENDADA (
    corridaAgendadaId INT AUTO_INCREMENT PRIMARY KEY,

    passageiroId INT NOT NULL,

    motoristaId INT NULL,

    pontoEncontroId INT NULL,

    origem VARCHAR(255) NOT NULL,

    destino VARCHAR(255) NOT NULL,

    dataHora DATETIME NOT NULL,

    status ENUM(
        'Agendada',
        'Confirmada',
        'EmAndamento',
        'Concluida',
        'Cancelada'
    ) NOT NULL DEFAULT 'Agendada',

    CONSTRAINT fk_corrida_agendada_passageiro
        FOREIGN KEY (passageiroId)
        REFERENCES PASSAGEIRO(passageiroId)
        ON UPDATE CASCADE,

    CONSTRAINT fk_corrida_agendada_motorista
        FOREIGN KEY (motoristaId)
        REFERENCES MOTORISTA(motoristaId)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT fk_corrida_agendada_ponto
        FOREIGN KEY (pontoEncontroId)
        REFERENCES PONTO_DE_ENCONTRO(pontoEncontroId)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);


-- =========================================================
-- 13. CORRIDA
-- =========================================================

CREATE TABLE CORRIDA (
    corridaId INT AUTO_INCREMENT PRIMARY KEY,

    passageiroId INT NOT NULL,

    motoristaId INT NULL,

    pontoOrigemId INT NULL,

    pontoEncontroId INT NULL,

    eventoId INT NULL,

    corridaAgendadaId INT NULL,

    origem VARCHAR(255) NOT NULL,

    latitudeOrigem DECIMAL(10,8) NOT NULL,

    longitudeOrigem DECIMAL(11,8) NOT NULL,

    destino VARCHAR(255) NOT NULL,

    latitudeDestino DECIMAL(10,8) NOT NULL,

    longitudeDestino DECIMAL(11,8) NOT NULL,

    distanciaKm DECIMAL(8,2),

    valorCorrida DECIMAL(10,2),

    beneficiarioNome VARCHAR(100),

    beneficiarioTelefone VARCHAR(20),

    tokenCompartilhamento VARCHAR(255),

    dataSolicitacao DATETIME
        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    dataInicio DATETIME NULL,

    dataFim DATETIME NULL,

    status ENUM(
        'Solicitada',
        'Aceita',
        'MotoristaChegando',
        'EmAndamento',
        'Finalizada',
        'Cancelada'
    ) NOT NULL DEFAULT 'Solicitada',

    motivoCancelamento VARCHAR(255),

    CONSTRAINT fk_corrida_passageiro
        FOREIGN KEY (passageiroId)
        REFERENCES PASSAGEIRO(passageiroId)
        ON UPDATE CASCADE,

    CONSTRAINT fk_corrida_motorista
        FOREIGN KEY (motoristaId)
        REFERENCES MOTORISTA(motoristaId)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT fk_corrida_ponto_interesse
        FOREIGN KEY (pontoOrigemId)
        REFERENCES PONTO_INTERESSE(pontoInteresseId)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT fk_corrida_ponto_encontro
        FOREIGN KEY (pontoEncontroId)
        REFERENCES PONTO_DE_ENCONTRO(pontoEncontroId)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT fk_corrida_evento
        FOREIGN KEY (eventoId)
        REFERENCES EVENTO(eventoId)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT fk_corrida_agendada
        FOREIGN KEY (corridaAgendadaId)
        REFERENCES CORRIDA_AGENDADA(corridaAgendadaId)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);


-- =========================================================
-- 14. PAGAMENTO
-- =========================================================

CREATE TABLE PAGAMENTO (
    pagamentoId INT AUTO_INCREMENT PRIMARY KEY,

    corridaId INT NOT NULL UNIQUE,

    valorPago DECIMAL(10,2) NOT NULL,

    formaPagamento ENUM(
        'Pix',
        'Dinheiro',
        'CartaoCredito',
        'CartaoDebito',
        'CarteiraDigital'
    ) NOT NULL,

    dataPagamento DATETIME NULL,

    statusPagamento ENUM(
        'Pendente',
        'Pago',
        'Cancelado'
    ) NOT NULL DEFAULT 'Pendente',

    CONSTRAINT fk_pagamento_corrida
        FOREIGN KEY (corridaId)
        REFERENCES CORRIDA(corridaId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 15. AVALIACAO
-- =========================================================

CREATE TABLE AVALIACAO (
    avaliacaoId INT AUTO_INCREMENT PRIMARY KEY,

    corridaId INT NOT NULL UNIQUE,

    passageiroId INT NOT NULL,

    motoristaId INT NOT NULL,

    nota TINYINT NOT NULL,

    comentario TEXT,

    dataAvaliacao DATETIME
        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT chk_avaliacao_nota
        CHECK (nota BETWEEN 1 AND 5),

    CONSTRAINT fk_avaliacao_corrida
        FOREIGN KEY (corridaId)
        REFERENCES CORRIDA(corridaId)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_avaliacao_passageiro
        FOREIGN KEY (passageiroId)
        REFERENCES PASSAGEIRO(passageiroId)
        ON UPDATE CASCADE,

    CONSTRAINT fk_avaliacao_motorista
        FOREIGN KEY (motoristaId)
        REFERENCES MOTORISTA(motoristaId)
        ON UPDATE CASCADE
);


-- =========================================================
-- 16. LOCALIZACAO
-- =========================================================

CREATE TABLE LOCALIZACAO (
    localizacaoId BIGINT AUTO_INCREMENT PRIMARY KEY,

    corridaId INT NOT NULL,

    motoristaId INT NOT NULL,

    latitude DECIMAL(10,8) NOT NULL,

    longitude DECIMAL(11,8) NOT NULL,

    dataHora DATETIME
        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_localizacao_corrida
        FOREIGN KEY (corridaId)
        REFERENCES CORRIDA(corridaId)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_localizacao_motorista
        FOREIGN KEY (motoristaId)
        REFERENCES MOTORISTA(motoristaId)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================================
-- 17. FAVORITO_MOTORISTA
-- =========================================================

CREATE TABLE FAVORITO_MOTORISTA (
    favoritoId INT AUTO_INCREMENT PRIMARY KEY,

    passageiroId INT NOT NULL,

    motoristaId INT NOT NULL,

    dataFavoritado DATETIME
        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_favorito_passageiro
        FOREIGN KEY (passageiroId)
        REFERENCES PASSAGEIRO(passageiroId)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_favorito_motorista
        FOREIGN KEY (motoristaId)
        REFERENCES MOTORISTA(motoristaId)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    -- Evita que o mesmo passageiro favorite
    -- o mesmo motorista várias vezes.
    CONSTRAINT uk_favorito_passageiro_motorista
        UNIQUE (passageiroId, motoristaId)
);


-- =========================================================
-- ÍNDICES
-- =========================================================

CREATE INDEX idx_telefone_usuario
ON TELEFONE(userId);

CREATE INDEX idx_endereco_usuario
ON ENDERECO(userId);

CREATE INDEX idx_endereco_favorito_usuario
ON ENDERECO_FAVORITO(userId);

CREATE INDEX idx_corrida_passageiro
ON CORRIDA(passageiroId);

CREATE INDEX idx_corrida_motorista
ON CORRIDA(motoristaId);

CREATE INDEX idx_corrida_status
ON CORRIDA(status);

CREATE INDEX idx_corrida_ponto_interesse
ON CORRIDA(pontoOrigemId);

CREATE INDEX idx_corrida_ponto_encontro
ON CORRIDA(pontoEncontroId);

CREATE INDEX idx_corrida_evento
ON CORRIDA(eventoId);

CREATE INDEX idx_corrida_agendada
ON CORRIDA(corridaAgendadaId);

CREATE INDEX idx_corrida_agendada_passageiro
ON CORRIDA_AGENDADA(passageiroId);

CREATE INDEX idx_corrida_agendada_motorista
ON CORRIDA_AGENDADA(motoristaId);

CREATE INDEX idx_ponto_interesse_cidade
ON PONTO_INTERESSE(cidade);

CREATE INDEX idx_ponto_interesse_categoria
ON PONTO_INTERESSE(categoria);

CREATE INDEX idx_evento_cidade
ON EVENTO(cidade);

CREATE INDEX idx_evento_data
ON EVENTO(dataInicio, dataFim);

CREATE INDEX idx_localizacao_corrida
ON LOCALIZACAO(corridaId);

CREATE INDEX idx_localizacao_motorista
ON LOCALIZACAO(motoristaId);

CREATE INDEX idx_favorito_passageiro
ON FAVORITO_MOTORISTA(passageiroId);

CREATE INDEX idx_favorito_motorista
ON FAVORITO_MOTORISTA(motoristaId);
